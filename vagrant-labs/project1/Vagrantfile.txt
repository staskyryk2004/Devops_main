Vagrant.configure("2") do |config|
  config.vm.box= "generic/ubuntu2004"

  config.vm.provider "qemu" do |qe|
  qe.arch = "aarch64"
    qe.cpu = "max"
    qe.machine = "virt,highmem=on"
    qe.advanced_network = true
  end

  config.vm.box_check_update = false

$enable_forwarding = <<-SHELL
  sudo sed -i 's?#net.ipv4.ip_forward=1/net.ipv4.ip_forward=1/g' /etc/sysctl.conf
  sudo sysctl -p
SHELL

config.vm.define "vm-a" do |a|
  a.vm.hostname = "vm-a"

  a.vm.network "private_network", ip:"172.16.0.10"

  a.vm.network "private_network", ip:"192.168.100.10"
  a.vm.provision "shell", inline:$enable_forwarding
end

config.vm.define "vm-b" do |b|
  b.vm.hostname = "vm-b"

  b.vm.network "private_network", ip:"192.168.101.20"

  b.vm.network "private_network", ip:"192.168.100.20"

  b.vm.provision "shell", inline:$enable_forwarding
  b.vm.provision "shell", inline:"sudo ip route del default dev eth0", run: "always"
end

config.vm.define "vm-c" do |c|
  c.vm.hostname = "vm-c"

  c.vm.network "private_network", ip:"192.168.102.30"

  c.vm.network "private_network", ip:"192.168.100.30"
  
  c.vm.provision "shell", inline:$enable_forwarding
 end
end
